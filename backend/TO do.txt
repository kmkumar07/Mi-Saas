import express from "express";
import fetch from "node-fetch";
import dotenv from "dotenv";
dotenv.config();

const app = express();
app.use(express.json());

// Cache entitlements in RAM for 5 minutes
const entitlementCache = new Map();

async function fetchEntitlements(tenantId, customerId) {
  const key = `${tenantId}:${customerId}`;
  
  if (entitlementCache.has(key)) {
    const { expires, data } = entitlementCache.get(key);
    if (Date.now() < expires) return data;
  }

  const res = await fetch(`${process.env.MINI_SAAS_URL}/entitlements/${tenantId}/${customerId}`, {
    headers: {
      "x-api-key": process.env.MINI_SAAS_KEY,
      "x-tenant-id": tenantId
    }
  });

  const data = await res.json();
  entitlementCache.set(key, {
    expires: Date.now() + 5 * 60 * 1000,
    data
  });

  return data;
}
async function requireFeature(featureCode) {
    return async (req, res, next) => {
      const tenantId = req.headers["x-tenant-id"];
      const customerId = req.headers["x-customer-id"];
  
      const entitlements = await fetchEntitlements(tenantId, customerId);
  
      if (!entitlements.features[featureCode]) {
        return res.status(403).json({
          message: "Upgrade to use this feature",
          requiredFeature: featureCode
        });
      }
  
      next();
    };
  }
